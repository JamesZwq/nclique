\section{The Structure of the Spectrum}
\label{sec:theory}

In this section, we prove the transfer theory that makes the spectrum computable without per-cell recomputation.
%
We first establish a closed-form floor for every cell (Theorem~\ref{thm:floor}), then a shadow bound linking adjacent cells (Theorem~\ref{thm:kk}), and combine them into the chain certificate (Theorem~\ref{thm:chain}); Theorem~\ref{thm:diagonal} gives the analogous transfer along the $s=r{+}1$ diagonal.

\stitle{The closed-form floor.}
Intuitively, the densest neighborhood an $r$-clique is guaranteed to have is the largest clique containing it, and that clique alone already forces a core value.

\begin{theorem}[Clique Floor]
\label{thm:floor}
For every $r$-clique $R$ and every cell $(r,s)$, $\corev{r,s}(R)\ \ge\ \binom{\cliqb(R)-r}{\,s-r\,}$.
\end{theorem}
\begin{proof}
If $\cliqb(R)<s$ the right side is $0$ and the claim is trivial.
%
Otherwise let $K$ be a clique containing $R$ with $|K|=\cliqb(R)$, and let $F$ be the family of all $s$-cliques inside $K$.
%
Every $r$-clique covered by $F$ lies in $K$ and has support exactly $\binom{\cliqb(R)-r}{s-r}$ in $F$, and $F$ covers $R$.
%
By Definition~\ref{def:core}, $\corev{r,s}(R)\ge\binom{\cliqb(R)-r}{s-r}$.
\end{proof}

We call an $r$-clique \textit{settled} at cell $s$ when Theorem~\ref{thm:floor} holds with equality.
%
The floor is a whole surface from one number: if $R$ is settled at every cell, its entire spectrum is $s\mapsto\binom{\cliqb(R)-r}{s-r}$, computed from $\cliqb(R)$ by arithmetic.
%
The difficulty is that a floor alone certifies nothing: the core value may exceed it, and equality cannot be recognized without an upper bound.

\stitle{The shadow bound.}
The missing upper bound comes from the neighboring cell.
%
For $a\ge 1$ and an integer $m\ge 0$, write $m=\binom{x}{a}$ for the unique real $x\ge a-1$, and define the \textit{shadow function} $g_a(m)=\binom{x}{a-1}$; this is the Lov\'{a}sz form~\cite{lovaszKK} of the Kruskal--Katona lower shadow bound~\cite{kruskal1963number, katona1968theorem}.

\begin{theorem}[Shadow Bound]
\label{thm:kk}
For every $r$-clique $R$ and every cell with $s\ge r+2$,
$\corev{r,s-1}(R)\ \ge\ g_{s-r}\big(\corev{r,s}(R)\big)$.
\end{theorem}
\begin{proof}
Let $k=\corev{r,s}(R)$, realized by the family $F$, and let $F'$ be the family of all $(s{-}1)$-cliques contained in members of $F$.
%
Every $r$-clique $R'$ covered by $F'$ lies in a member of $F$; its extension family $E(R')=\{S\setminus R' : S\in F,\ R'\subseteq S\}$ is $(s{-}r)$-uniform with $|E(R')|=\mathrm{sup}_F(R')\ge k$.
%
Every $(s{-}1{-}r)$-set in the lower shadow of $E(R')$ extends $R'$ to an $(s{-}1)$-clique inside a member of $F$, so $\mathrm{sup}_{F'}(R')\ge g_{s-r}(|E(R')|)\ge g_{s-r}(k)$ by Kruskal--Katona and the monotonicity of $g_{s-r}$.
%
$F'$ covers $R$ because $s-r\ge 2$ allows dropping one vertex of $S\setminus R$ for any $S\in F$ containing $R$.
%
By Definition~\ref{def:core}, the claim follows.
\end{proof}

\begin{remark}
The shadow bound has zero slack at binomial values: if $k=\binom{y}{a}$ for an integer $y$, then $g_a(k)=\binom{y}{a-1}$ exactly.
%
On a clique, the chain of floor values $\binom{c-r}{s-r}$ is therefore $g$-tight at every cell; this exactness is what turns the bound into a certificate.
\end{remark}

\stitle{The chain certificate.}
Combining the floor from below with the shadow bound from above squeezes the core value exactly, using only the previous cell.

\begin{theorem}[Chain Certificate]
\label{thm:chain}
Let $R$ be an $r$-clique, $c=\cliqb(R)$, and $s\ge r+2$.
%
If $\corev{r,s-1}(R)=\binom{c-r}{s-1-r}$, then $\corev{r,s}(R)=\binom{c-r}{s-r}$.
\end{theorem}
\begin{proof}
The lower bound is Theorem~\ref{thm:floor}.
%
If $c<s$, no $s$-clique contains $R$ and both sides are $0$.
%
Otherwise suppose $\corev{r,s}(R)\ge\binom{c-r}{s-r}+1$ and let $a=s-r\ge 2$.
%
By Theorem~\ref{thm:kk}, $\corev{r,s-1}(R)\ge g_a\big(\binom{c-r}{a}+1\big)$.
%
The real solution $x(m)$ of $\binom{x}{a}=m$ is strictly increasing in $m$, and $x\mapsto\binom{x}{a-1}$ is strictly increasing for $x>a-1$; at the integer point $x=c-r\ge a$ this gives $g_a\big(\binom{c-r}{a}+1\big)>\binom{c-r}{a-1}=\corev{r,s-1}(R)$, a contradiction.
\end{proof}

\begin{example}
\label{exa:chain}
Consider the triangle $R=\{v_3,v_4,v_5\}$ in Figure~\ref{fig:running}, with $\cliqb(R)=5$.
%
Example~\ref{exa:core} gives $\corev{3,4}(R)=2=\binom{2}{1}$, so the hypothesis of Theorem~\ref{thm:chain} holds at $s=5$.
%
Therefore $\corev{3,5}(R)=\binom{2}{2}=1$ with no computation at cell $(3,5)$ at all.
\end{example}

\begin{remark}
Certification is absorbing: if $R$ is certified at cell $s$, its new core value again equals the clique value, so the hypothesis of Theorem~\ref{thm:chain} holds at $s+1$ with the same $c$.
%
One equality check therefore settles not one cell but the entire remaining spectrum of $R$, and the check itself is one integer comparison.
\end{remark}

\stitle{Why weaker transfers fail.}
It is natural to ask whether the same squeeze exists between rows, that is, from order $r$ to order $r{+}1$ at a fixed $s$.
%
The natural bound does exist, but it cannot certify.

\begin{theorem}[Subclique Bound]
\label{thm:subclique}
For every $(r{+}1)$-clique $R^{+}$ and every cell $(r{+}1,s)$,
$\corev{r+1,s}(R^{+})\ \le\ \min_{R\subset R^{+}} \corev{r,s}(R)$,
where $R$ ranges over the $r$-subcliques of $R^{+}$.
\end{theorem}
\begin{proof}
Let $F$ realize $k=\corev{r+1,s}(R^{+})$ and let $R'$ be any $r$-clique covered by $F$, say $R'\subseteq S\in F$.
%
Picking $u\in S\setminus R'$ makes $R'\cup\{u\}$ an $(r{+}1)$-clique covered by $F$, so it lies in at least $k$ members of $F$, each of which also contains $R'$.
%
Hence every $r$-clique covered by $F$ has support at least $k$, and $F$ witnesses $\corev{r,s}(R')\ge k$; applying this to the $r$-subcliques of $R^{+}$ gives the claim.
\end{proof}

\begin{remark}
On a $c$-clique, Theorem~\ref{thm:subclique} bounds $\binom{c-r-1}{s-r-1}$ by $\binom{c-r}{s-r}$, which is strictly larger: the bound has slack even in the best case, so no floor can meet it and no certificate follows.
%
The contrast identifies exactly what makes Theorem~\ref{thm:chain} work: the shadow bound is tight at binomial values, the subclique bound is not.
\end{remark}

\stitle{The diagonal.}
One cross-row transfer does achieve zero slack: along the diagonal of cells $s=r{+}1$, where the classical inequality between the truss number and the core number~\cite{cohen2008trusses} is the case $r=1$.

\begin{theorem}[Diagonal Transfer]
\label{thm:diagonal}
For every $(r{+}1)$-clique $R^{+}$,
$\corev{r+1,r+2}(R^{+})\ \le\ \Big(\min_{R\subset R^{+}} \corev{r,r+1}(R)\Big)\ -\ 1$.
\end{theorem}
\begin{proof}
Let $F$ be a family of $(r{+}2)$-cliques realizing $k=\corev{r+1,r+2}(R^{+})$, and let $F'$ be the family of all $(r{+}1)$-cliques contained in members of $F$.
%
Let $R'$ be any $r$-clique covered by $F'$ and pick $W=R'\cup\{u\}\in F'$.
%
$W$ is an $(r{+}1)$-clique covered by $F$, so at least $k$ distinct members $S_1,\dots,S_k$ of $F$ contain $W$, and each $S_i=W\cup\{w_i\}$ with the $w_i$ pairwise distinct and outside $W$.
%
Now $F'$ contains the $k{+}1$ distinct $(r{+}1)$-cliques $R'\cup\{u\}$ and $R'\cup\{w_i\}$ for $i=1,\dots,k$, the latter because each is an $(r{+}1)$-subset of the clique $S_i$.
%
Hence $\mathrm{sup}_{F'}(R')\ge k+1$, and $F'$ witnesses $\corev{r,r+1}(R')\ge k+1$ for every covered $r$-clique, in particular every $r$-subclique of $R^{+}$.
\end{proof}

\begin{corollary}[Diagonal Certificate]
\label{cor:diagcert}
If $\min_{R\subset R^{+}}\corev{r,r+1}(R)=\cliqb(R^{+})-r$, then $\corev{r+1,r+2}(R^{+})=\cliqb(R^{+})-r-1$ exactly.
\end{corollary}

\begin{remark}
On a $c$-clique both sides of Theorem~\ref{thm:diagonal} meet: the left side is $c-r-1$ and the right side is $(c-r)-1$.
%
Across $156$ million $4$-cliques on three real graphs, the inequality holds with zero violations, and it is an equality for $100\%$ of the $4$-cliques on the collaboration graphs and $23.4\%$ on a social graph (Section~\ref{sec:exp}).
\end{remark}

\stitle{From certificates to an algorithm.}
Theorems~\ref{thm:floor} through~\ref{thm:chain} reduce the spectrum to three ingredients: one number $\cliqb(\cdot)$ per pattern, one exactly computed boundary cell, and, per subsequent cell, one integer comparison per pattern plus the small uncertified residue.
%
However, peeling the residue exactly while the certified mass is never recomputed remains a technical challenge; Section~\ref{sec:algo} resolves it.
