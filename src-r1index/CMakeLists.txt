cmake_minimum_required(VERSION 3.20)
project(r1index LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(chain_index_tool chain_index_tool.cpp)
target_include_directories(chain_index_tool PRIVATE ${CMAKE_CURRENT_SOURCE_DIR} /opt/homebrew/include)
target_compile_options(chain_index_tool PRIVATE -Wall -Wextra -Wpedantic)
option(SANITIZE "Enable address and undefined behavior sanitizers" OFF)
if(SANITIZE)
  target_compile_options(chain_index_tool PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(chain_index_tool PRIVATE -fsanitize=address,undefined)
endif()
