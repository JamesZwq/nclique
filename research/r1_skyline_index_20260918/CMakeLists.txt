cmake_minimum_required(VERSION 3.20)
project(r1_skyline_count LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(count count.cpp)
target_include_directories(count PRIVATE /opt/homebrew/include)
target_compile_options(count PRIVATE -Wall -Wextra -Wpedantic)
option(SANITIZE "Enable address and undefined behavior sanitizers" OFF)
if(SANITIZE)
  target_compile_options(count PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(count PRIVATE -fsanitize=address,undefined)
endif()
