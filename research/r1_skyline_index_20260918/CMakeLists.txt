cmake_minimum_required(VERSION 3.20)
project(r1_skyline_index LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(count count.cpp)
add_executable(index index.cpp)
add_executable(chains chains.cpp)
foreach(target count index chains)
  target_include_directories(${target} PRIVATE /opt/homebrew/include)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
endforeach()
option(SANITIZE "Enable address and undefined behavior sanitizers" OFF)
if(SANITIZE)
  foreach(target count index chains)
    target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${target} PRIVATE -fsanitize=address,undefined)
  endforeach()
endif()
