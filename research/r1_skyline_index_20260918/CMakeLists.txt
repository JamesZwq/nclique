cmake_minimum_required(VERSION 3.20)
project(r1_skyline_index LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(count count.cpp)
add_executable(index stages/index.cpp)
add_executable(chains stages/chains.cpp)
add_executable(chain_index_tool chain_index_tool.cpp)
add_executable(strees_dump strees_dump.cpp)
add_executable(query_profile query_profile.cpp)
add_executable(tail_check tail_check.cpp)
# 64-bit row index (member offsets, row ids, reverse codes) for clique trees past 2^32 incidences or 2^30 rows
add_executable(chain_index_tool_wide chain_index_tool.cpp)
target_compile_definitions(chain_index_tool_wide PRIVATE TERMINAL_WIDE)
foreach(target count index chains chain_index_tool chain_index_tool_wide strees_dump query_profile tail_check)
  target_include_directories(${target} PRIVATE /opt/homebrew/include)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
endforeach()
option(SANITIZE "Enable address and undefined behavior sanitizers" OFF)
if(SANITIZE)
  foreach(target count index chains chain_index_tool)
    target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${target} PRIVATE -fsanitize=address,undefined)
  endforeach()
endif()
