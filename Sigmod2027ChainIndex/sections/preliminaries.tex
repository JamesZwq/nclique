\section{Preliminaries}\label{sec:prelim}
% 2026-09-21 house shape: prose notation, one definition with its example on Figure 1, the two queries, a two-sentence problem statement.

\stitle{Graphs and cliques.}
% 2026-09-22 prose pass: words before symbols; the clique-number example says why.
We consider a simple undirected graph $G=(V,E)$ with $n$ vertices.
%
An \textit{$s$-clique} is a set of $s$ pairwise adjacent vertices.
%
A $2$-clique is an edge, and a $3$-clique is a triangle.
%
The \textit{clique number} of a vertex $v$, denoted as $\omg{v}$, is the largest size of a clique containing $v$.
%
A subset of a clique is a clique, so $v$ lies in an $s$-clique for every size $s$ from $2$ to $\omg{v}$ and for no larger size.
%
In Figure~\ref{fig:example}, the clique number of $a_1$ is $5$, since $A$ has five vertices.
%
The clique number of $x$ is $4$ because of $B$.
%
That of $u_1$ is $3$: it lies in the triangles $\{x,u_1,w_j\}$ for $j=1,2,3$ and in no $4$-clique, since no two of $u_1,u_2,u_3$ are adjacent.

\stitle{Nuclei.}
% 2026-09-22 cold-read fixes: the "1" of (1,s) explained, maximal glossed, the connectivity property named.
The community model of this paper is the $(1,s)$-nucleus of Sariy\"uce et al.~\cite{firstNucleus}.
%
The $1$ says that the unit is a single vertex, and the $s$ is the size of the cliques counted around it.
%
We write a nucleus of clique size $s$ at level $k$ as an $\Nuc{s}{k}$.
\begin{definition}[$(s,k)$-nucleus]\label{def:nucleus}
For a clique size $s\ge2$ and a level $k\ge1$, an $\Nuc{s}{k}$ is a set $N\subseteq V$ with two properties, to which no vertex can be added without losing one of them.
%
Every vertex of $N$ lies in at least $k$ cliques of $s$ vertices contained in $N$.
%
Any two vertices of $N$ are joined by a sequence of $s$-cliques contained in $N$: the first clique contains one of the two vertices, the last contains the other, and consecutive cliques share a vertex.
%
We say that $N$ is connected through its $s$-cliques.
\end{definition}
For $s=2$ the cliques are edges, and a $(2,k)$-nucleus is a connected $k$-core: a maximal connected subgraph in which every vertex has at least $k$ neighbours.
%
Two nuclei of the same size $s$ and level $k$ are disjoint.
%
If they shared a vertex, their union would have both properties, the two clique sequences being joined at the shared vertex, so each of them could be extended.
%
The \textit{core value} of $v$ at size $s$, denoted as $\core{s}{v}$, is the largest $k$ such that $v$ belongs to an $\Nuc{s}{k}$.
%
It is $0$ if $v$ lies in no $s$-clique.
%
The \textit{community} of $v$ at $(s,k)$, for $1\le k\le\core{s}{v}$, is the $\Nuc{s}{k}$ containing $v$.
%
Such a nucleus exists.
%
The $\Nuc{s}{\core{s}{v}}$ containing $v$ has the two properties at level $k$ as well, since $k\le\core{s}{v}$, and adding vertices to it until neither property can be kept gives an $\Nuc{s}{k}$ containing $v$.
%
This nucleus is unique, since nuclei of one size and level are disjoint.
\begin{example}\label{ex:nucleus}
Consider Figure~\ref{fig:example} at size $3$.
%
As in Example~\ref{ex:intro}, $\{u_1,\dots,w_3\}$ denotes the six vertices $u_1,u_2,u_3,w_1,w_2,w_3$.
%
A triangle of $A$ through $a_i$ is $a_i$ with two of the other four vertices, so $a_i$ lies in $\binom{4}{2}=6$ triangles of $A$.
%
It lies in no other triangle, since the edge $a_5b_3$ lies in none.
%
$A$ is therefore a $(3,6)$-nucleus, and $\core{3}{a_i}=6$.
%
Each of $b_1,b_2,b_3$ lies in three triangles of $B$.
%
Each $u_i$ lies in the three triangles $\{u_i,w_j,x\}$ for $j=1,2,3$, and so does each $w_j$.
%
The vertex $x$ lies in the three triangles of $B$ and in the nine triangles $\{x,u_i,w_j\}$, twelve in all.
%
The set $N=B\cup\{u_1,\dots,w_3\}$ is a $(3,3)$-nucleus.
%
Every vertex of $N$ lies in at least three of its triangles.
%
Every triangle of $N$ except $\{b_1,b_2,b_3\}$ contains $x$, and that one shares $b_1$ with $\{x,b_1,b_2\}$, so $N$ is connected through its triangles.
%
No vertex of $A$ can be added, since the edge $a_5b_3$ lies in no triangle.
%
No vertex of $N$ has core value $4$ at size $3$.
%
A $(3,4)$-nucleus needs every vertex in four of its triangles, so it contains none of $u_1,\dots,w_3$, and without them a vertex of $B$ lies in the three triangles of $B$ only.
%
So $\core{3}{v}=3$ for the ten vertices of $N$, four of $B$ and six of the bipartite graph.
%
At size $2$ the core value of a vertex is the largest $k$ for which it lies in a $k$-core.
%
Every vertex has core value at least $3$, since the smallest degree of the graph is $3$.
%
$\core{2}{a_i}=4$: each of $a_1,\dots,a_4$ has four neighbours, so no $5$-core exists, and $A$ is a $4$-core.
%
$\core{2}{b_i}=3$: $b_1$ and $b_2$ have three neighbours, and once they are removed $b_3$ keeps two, so none of the three lies in a $4$-core.
%
$\core{2}{v}=4$ for $x$ and for the six vertices $u_1,\dots,w_3$: inside $\{x,u_1,\dots,w_3\}$, $x$ has six neighbours and every other vertex has four.
% trace 2026-09-22 (make_concept.py, brute force): kappa_3 = 6 on A, 3 on N; kappa_2 = 4 on A and on {x,u,w}, 3 on b_i; kappa_4 = 4 on A, 1 on B; kappa_5 = 1 on A.
%
At size $4$ only $A$ and $B$ have a nonzero value, $\core{4}{a_i}=\binom{4}{3}=4$ and $\core{4}{v}=1$ for the vertices of $B$.
%
At size $5$, $\core{5}{a_i}=1$.
\end{example}

\stitle{Queries.}
An index for this problem answers two queries, for every size $s$.
%
A \textit{value query} $(v,s)$ returns $\core{s}{v}$.
%
A \textit{community query} $(v,s,k)$ with $1\le k\le\core{s}{v}$ returns the vertex set of the community of $v$ at $(s,k)$.
%
Whether a vertex belongs to a community follows from the community query.
%
The list of the communities of $v$ at one size, one per level from $1$ to $\core{s}{v}$, follows from community queries as well.
%
Neither is a separate query.

\stitle{Problem statement.}
Given a graph $G$, we aim to build an index that answers value and community queries for every clique size $s\ge2$.
%
The index should store each vertex once and return every community as a few ranges of consecutive vertex ids.
