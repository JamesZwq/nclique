\stitle{One clique tree for every size.}
% 2026-09-24 user: tell the static clique tree story (idea of the unpublished r = 1 paper, used here, not cited): one tree,
% never edited, serves the peel of every size and the tree of every size.  Definition, count lemma, removal lemma.
The clique counts of every size come from one pivoting clique tree~\cite{Pivoter}.
\begin{definition}[clique tree]\label{def:cliquetree}
A \textit{clique tree} of $G$ is a set of paths.
%
A path $P$ carries two disjoint vertex sets, a nonempty \textit{mandatory set} $H$ and an \textit{optional set} $Q$, whose union is a clique.
%
At size $s$ the path represents the $s$-cliques $H\cup Z$ for every $Z\subseteq Q$ with $|Z|=s-|H|$, and every $s$-clique of $G$ is represented by exactly one path.
\end{definition}
A path represents cliques at the sizes from $|H|$ to $|H|+|Q|$, so one clique tree serves every size.
%
For a vertex $v$ and a vertex set $W$, write $c_s(v,W)$ for the clique count of $v$ in $W$, the number of $s$-cliques that contain $v$ and lie inside $W$.
\begin{lemma}[clique count]\label{lem:count}
For $v\in W$, $c_s(v,W)$ is the sum, over the paths through $v$ whose mandatory set lies in $W$, of $\binomc{|Q\cap W|-r}{s-|H|-r}$, where $r=1$ if $v\in Q$ and $r=0$ otherwise.
\end{lemma}
\begin{proof}
An $s$-clique through $v$ inside $W$ is represented by one path $P$ through $v$ as $H\cup Z$, and it lies in $W$ exactly when $H\subseteq W$ and $Z\subseteq Q\cap W$.
%
If $v\in H$, any $s-|H|$ vertices of $Q\cap W$ form $Z$, and if $v\in Q$, $Z$ contains $v$ and $s-|H|-1$ of the other vertices of $Q\cap W$.
\end{proof}
The peel of a size removes vertices from $W$ one at a time, and the next lemma gives the change of a count at a removal.
\begin{lemma}[removal]\label{lem:removal}
Let $u\ne v$ lie in $W$.
%
Then $c_s(u,W)-c_s(u,W\setminus\{v\})$ is the sum, over the paths through both $u$ and $v$ whose mandatory set lies in $W$, of $\binomc{|Q\cap W|-r}{s-|H|-r}$, where $r$ is the number of the two vertices $u$ and $v$ that lie in $Q$.
\end{lemma}
\begin{proof}
The cliques counted in $c_s(u,W)$ and not in $c_s(u,W\setminus\{v\})$ are those that contain $v$ as well.
%
By the argument of Lemma~\ref{lem:count}, a path through both vertices represents $\binomc{|Q\cap W|-r}{s-|H|-r}$ of them, and a path without $v$ represents none.
\end{proof}
A path thus depends on $W$ only through whether its mandatory set lies in $W$ and through $|Q\cap W|$.
%
Removing a mandatory vertex ends the path's contribution, and removing an optional vertex lowers $|Q\cap W|$ by one.
%
The peel therefore keeps for every path one \textit{counter}, the number $q(P)=|Q\cap W|$ of its optional vertices not yet removed while its mandatory set is intact, and never edits a path, so the clique tree is \textit{static}.

\stitle{Change-only peel with settled vertices.}
The peel of a size counts cliques for every vertex, but Theorem~\ref{thm:tail} already gives many of the values that the peel computes.
%
A vertex $v$ with $\omg{v}\ge s$ is \textit{settled} at size $s$ when $s>\sig{v}$, that is, when its value at size $s-1$ is already its floor.
%
By Theorem~\ref{thm:tail}(b), its value at size $s$ is then the floor $\binomc{\omg{v}-1}{s-1}$.
%
The other vertices, the \textit{unsettled} ones, are peeled.
%
In the peel below, the \textit{key} of a settled vertex is its value, and the key of an unsettled vertex is its clique count among the vertices not yet removed.
%
At size $2$, the first size peeled, no vertex is settled.
\begin{theorem}[peel with settled vertices]\label{thm:settled}
Let every vertex $v$ with $\omg{v}\ge s$ be either settled, with key $\core{s}{v}$, or unsettled, with key its clique count.
%
Remove these vertices one at a time, each time one of smallest key, and give each the largest key seen at a removal so far.
%
Every such vertex receives its core value $\core{s}{v}$.
\end{theorem}
\begin{proof}
% 2026-09-23 proof check (codex exhaustive to 6 vertices with every bound, settled set and tie order; Fable 42,408 random peels): true; the maximality step, the empty case and the settled vertex's membership in D_k are now written.
Fix a level $k\ge1$, and let $D_k$ be the vertices of value at least $k$, the union of the cores of size $s$ at level $k$.
%
Every vertex of $D_k$ lies in at least $k$ $s$-cliques inside $D_k$.
%
Conversely, a set whose every vertex lies in $k$ or more $s$-cliques inside it lies inside $D_k$: each part connected through these cliques has both properties of Definition~\ref{def:nucleus} at level $k$.
%
Until a vertex of $D_k$ is removed, every vertex of $D_k$ has key at least $k$, since its value, or else its count, is at least $k$.
%
The first vertex of $D_k$ to be removed thus has key at least $k$, and a removal with key at least $k$ occurs whenever $D_k$ is not empty.
%
Just before the first removal with key at least $k$, the remaining vertices, call them $R$, include all of $D_k$.
%
Every vertex of $R$ has key at least $k$, and an unsettled one lies in at least $k$ $s$-cliques inside $R$ because its key is its count.
%
A settled one has value at least $k$ and lies in $D_k$, where it lies in at least $k$ $s$-cliques, all inside $R$.
%
By the converse, $R=D_k$, so exactly the vertices of $D_k$ receive a value of at least $k$, and every vertex receives $\core{s}{v}$.
\end{proof}
By Lemma~\ref{lem:removal}, a removal changes only the counts of the vertices that share a path with the removed vertex, and the peel never reads the count of a settled vertex.
%
Procedure \kw{Peel} of Algorithm~\ref{alg:peellevels} is this peel.
%
It keys the settled vertices by their floors, sets the counters, and counts the others by Lemma~\ref{lem:count} (Lines~\ref{ln:p-init}--\ref{ln:p-count}).
%
It removes a vertex of smallest key and gives it the largest key seen so far (Lines~\ref{ln:p-min}--\ref{ln:p-give}).
%
It then visits the paths through the removed vertex whose mandatory set was inside $A$ (Line~\ref{ln:p-paths}).
%
On each, it subtracts from every unsettled vertex still in $A$ the loss of Lemma~\ref{lem:removal}, the only change the removal causes, so the peel updates only what changes (Line~\ref{ln:p-update}).
%
If the removed vertex is optional on the path, the counter drops by one (Line~\ref{ln:p-counter}), and a path whose mandatory set lost a vertex is never visited again.
\begin{example}\label{ex:settled}
% trace 2026-09-23: tools/example_check.py (settled and unsettled per size, 12 triangles through the unsettled vertices, all removed at level 3).
Consider Figure~\ref{fig:example} at size $3$.
%
The vertices $a_1,\dots,a_5$ and $b_1,b_2,b_3$ have their floors $4$ and $3$ as values at size $2$ (Example~\ref{ex:tail}), and they are settled with the floors $\binom{4}{2}=6$ and $\binom{3}{2}=3$.
%
The value of $x$ and of the six vertices $u_1,\dots,w_3$ at size $2$ is $4$, above their floors $3$ and $2$, and these seven vertices are unsettled.
%
The peel counts triangles only for them: $x$ lies in $12$, nine with one of $u_1,u_2,u_3$ and one of $w_1,w_2,w_3$, and three inside $B$, and each of the other six lies in $3$.
%
% 2026-09-23 proof check: the order among equal keys is free; the sentences hold for every order.
The smallest keys are $3$, of $u_1,\dots,w_3$ and of the settled $b_1,b_2,b_3$, and these nine vertices leave at level $3$.
%
Once $u_1,\dots,w_3$ are gone, $x$ lies in at most the three triangles of $B$, fewer if some $b_i$ left first, and it leaves at level $3$ as well.
%
No triangle joins $a_1,\dots,a_5$ to an unsettled vertex, and their removals change no key.
%
At size $4$, $x$ is settled, since its value $3$ at size $3$ equals its floor $\binom{3}{2}$ there, and the $a_i$ and $b_i$ stay settled.
\end{example}

\stitle{The tree of a size.}
% 2026-09-24: the tree of a size from merge events of the static clique tree (the idea of the r = 1 paper's hierarchy step).
Once the values of size $s$ are known, the tree $\Tree{s}$ is read off the static clique tree without listing a clique.
%
A clique of a path enters the cores at the smallest value among its vertices.
%
For a path with $|H|\le s\le|H|+|Q|$, let $\eta=s-|H|$, let $h$ be the smallest value of size $s$ on $H$, and order $Q$ as $p_1,\dots,p_m$ by nonincreasing value, where $m=|Q|$ and $\core{s}{p_0}=\infty$.
%
The path makes one \textit{event} for its first clique, $H\cup\{p_1,\dots,p_\eta\}$ at level $\min(h,\core{s}{p_\eta})$, and, if $\eta\ge1$, one event $\{p_1,p_i\}$ at level $\min(h,\core{s}{p_i})$ for every $i>\eta$ (for $\eta=0$ the only event is $H$ at level $h$).
\begin{lemma}[events]\label{lem:events}
For every level $k\ge1$, joining the vertices of each event of level at least $k$ gives as components exactly the cores of size $s$ at level $k$.
\end{lemma}
\begin{proof}
An event of level at least $k$ joins vertices of value at least $k$ that are connected through $s$-cliques of such vertices.
%
The first event is a clique, and for $i>\eta$ the vertices $p_1$ and $p_i$ lie together in the clique $H\cup\{p_1,\dots,p_{\eta-1},p_i\}$ when $\eta\ge2$, whose values are at least $\min(h,\core{s}{p_i})$ by the order of $Q$, or share the cliques $H\cup\{p_1\}$ and $H\cup\{p_i\}$ when $\eta=1$.
%
Conversely, let $C=H\cup Z$ be an $s$-clique of a path whose vertices all have value at least $k$.
%
Then $h\ge k$, and the $\eta$ vertices of $Z$ have values at least $k$, so $\core{s}{p_\eta}\ge k$ and the first event has level at least $k$.
%
Every vertex $p_i$ of $Z$ with $i>\eta$ joins $p_1$ in an event of level $\min(h,\core{s}{p_i})\ge k$, so $C$ lies in one component.
%
The components are therefore the parts of the vertices of value at least $k$ connected through their $s$-cliques, the cores at level $k$.
\end{proof}
Procedure \kw{Hierarchy} of Algorithm~\ref{alg:peellevels} collects the events (Lines~\ref{ln:l-events}--\ref{ln:l-pairs}) and processes them by decreasing level with a union-find structure (Lines~\ref{ln:l-levels}--\ref{ln:l-union}).
%
By Lemma~\ref{lem:events}, after the events of level $k$ every component that changed is a canonical node of top $k$, and the nodes it absorbed become its children (Line~\ref{ln:l-node}).
%
Every vertex of value $k$ receives the node of top $k$ containing it as its own node (Line~\ref{ln:l-own}).

\begin{algorithm}[t]
\caption{\kw{Peel} and \kw{Hierarchy} at size $s$.}\label{alg:peellevels}
\SetAlgoLined\DontPrintSemicolon
\SetKwProg{Proc}{Procedure}{}{}
\SetKwFunction{Peel}{Peel}\SetKwFunction{Hierarchy}{Hierarchy}
\Proc{\Peel{$s,S$}}{
  $A\leftarrow\{v:\omg{v}\ge s\}$; $\ell\leftarrow0$\;\label{ln:p-a}
  $\mathrm{key}(v)\leftarrow\binomc{\omg{v}-1}{s-1}$, for every $v\in S$\;\label{ln:p-init}
  $q(P)\leftarrow|Q\cap A|$, for every path $P$\;
  $\mathrm{key}(v)\leftarrow c_s(v,A)$, for every $v\in A\setminus S$\;\label{ln:p-count}
  \While{$A\ne\emptyset$}{
    $v\leftarrow$ a vertex of $A$ with the smallest key\;\label{ln:p-min}
    $A\leftarrow A\setminus\{v\}$; $\ell\leftarrow\max(\ell,\mathrm{key}(v))$; $\core{s}{v}\leftarrow\ell$\;\label{ln:p-give}
    \ForEach{path $P$ through $v$ with $H\subseteq A\cup\{v\}$}{\label{ln:p-paths}
      $\mathrm{key}(u)\leftarrow\mathrm{key}(u)-\binomc{q(P)-r}{s-|H|-r}$, for $u\in(H\cup Q)\cap A\setminus S$, $r=|\{u,v\}\cap Q|$\;\label{ln:p-update}
      \lIf{$v\in Q$}{$q(P)\leftarrow q(P)-1$}\label{ln:p-counter}
    }
  }
  \Return $\core{s}{\cdot}$\;
}
\Proc{\Hierarchy{$s$}}{
  $E\leftarrow\emptyset$; $\Tree{s}\leftarrow\emptyset$\;
  \ForEach{path with $|H|\le s\le|H|+|Q|$}{\label{ln:l-events}
    add $H\cup\{p_1,\dots,p_\eta\}$ at level $\min(h,\core{s}{p_\eta})$ to $E$\;\label{ln:l-first}
    \lIf{$\eta\ge1$}{add $\{p_1,p_i\}$ at level $\min(h,\core{s}{p_i})$ to $E$, for $\eta<i\le m$}\label{ln:l-pairs}
  }
  \ForEach{level $k\ge1$ of $E$, in decreasing order}{\label{ln:l-levels}
    join the vertices of every event of level $k$\;\label{ln:l-union}
    add every component that changed to $\Tree{s}$ as a node of top $k$, with the nodes it absorbed as children\;\label{ln:l-node}
    $\own{s}{v}\leftarrow$ the node of top $k$ containing $v$, for $v$ with $\core{s}{v}=k$\;\label{ln:l-own}
  }
  \Return $(\Tree{s},X_s)$\;
}
\end{algorithm}

\stitle{Algorithm.}
% 2026-09-24: the walk follows the abstract float; the layout steps point to Section 5, where they are defined.
Algorithm~\ref{alg:build} builds the clique tree once (Line~\ref{ln:b-tree}), and by Lemma~\ref{lem:removal} no later step edits it; it starts from one class holding every vertex (Line~\ref{ln:b-part}).
%
For every size (Line~\ref{ln:b-sizes}) it computes the values (Step 1) and the tree with the refinement of the classes (Step 2), and after the last size it lays out the index (Step 3).
%
\emph{During Step 1}, the algorithm collects the vertices settled at size $s$ (Line~\ref{ln:b-settle}).
%
Based on Theorem~\ref{thm:settled}, \kw{Peel} keys them by their floors and counts cliques only for the other vertices (Line~\ref{ln:b-peel}).
%
\emph{During Step 2}, \kw{Hierarchy} builds $\Tree{s}$ and the own nodes of size $s$ from the same clique tree (Line~\ref{ln:b-hier}).
%
The algorithm then splits every class by the own node of size $s$, where $\own{s}{v}=\bot$ for a vertex of value $0$ (Line~\ref{ln:b-refine}).
%
After the last size, two vertices share a class exactly when they have the same own node at every size, so by Theorem~\ref{thm:chains} the classes are the chains.
%
Figure~\ref{fig:pipeline} draws this refinement as a trie, one level per size.
%
\emph{During Step 3}, the algorithm ranks the chains by the keys of Section~\ref{sec:order} and gives the vertices their labels (Line~\ref{ln:b-rank}).
%
It builds the run array and the entries of every size as in Section~\ref{sec:runs} (Line~\ref{ln:b-runs}).
%
It stores for every chain its clique number, its tail start, its residues and its trajectory, as in Section~\ref{sec:values} (Line~\ref{ln:b-values}).

It makes the event $\{a_1,a_2\}$ and the events $\{a_2,a_i\}$ for $i\ge3$, all at level $4$, and these join $A$.

The events of level $4$ join $\{x,u_1,\dots,w_3\}$ in the same way, and the two components are the nodes of top $4$.

\stitle{Cost.}
% 2026-09-24: bounds in terms of the incidences of the paths of a size, from the three lemmas.
Let $I_s$ be the total size of the paths with $|H|\le s\le|H|+|Q|$.
%
\kw{Peel} reads each such path once to count (Lemma~\ref{lem:count}) and once per removed vertex of the path while its mandatory set is intact (Lemma~\ref{lem:removal}).
%
\kw{Hierarchy} makes at most $I_s$ events and at most $I_s$ unions and sorts them in $O(I_s\log I_s)$ time.
%
The refinement adds one step per vertex and size, and the layout one pass per size over the chains.
%
The working space is the clique tree and the graph, plus the values of two consecutive sizes, the partition, and the trees of the sizes with one node per canonical node.
