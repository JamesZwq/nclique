\section{The Chain Index}\label{sec:index}
% 2026-09-21 house shape and cadence pass: overview + Figure 2; each block with its property, theorem, proof, example on
% Figures 1 and 2; the values block holds the certified tail; the size block closes with the accounting.

This section answers the second difficulty of Section~\ref{sec:intro}: how to return any community of any size as a few ranges of memory.
%
The index has three blocks, all flat arrays: the vertex-to-chain map, one record per chain, and for every size one tree over the chains with a run array.
%
Figure~\ref{fig:layout} shows the three blocks for the running example.
%
We describe the map and its labels (Section~\ref{sec:labels}), the order of the chains (Theorem~\ref{thm:onerange}), the run arrays (Theorem~\ref{thm:retrieval}), the values (Theorem~\ref{thm:tail}), and the size of the whole (Section~\ref{sec:size}).

\subsection{Aligned labels}\label{sec:labels}
Chains partition $V$ (Section~\ref{sec:chains}).
%
The vertices can be relabelled so that every chain is one interval of labels: chain $c$ owns $[\mathrm{start}(c),\mathrm{start}(c+1))$.
%
The map from a vertex to its chain is then a bitmap with a one at every chain start, and the last chain ends at label $n$.
%
The chain of $v$ is $\mathrm{rank}_1(v)-1$, where $\mathrm{rank}_1(v)$ is the number of ones at positions $0$ to $v$ of the bitmap.
%
A directory stores the number of ones before every block of $64$ labels, so a rank is one directory read and one population count of a $64$-bit word.
%
The map costs $1$ bit per vertex for the bitmap and $1/16$ byte per vertex for the directory, plus the first label of every chain.
%
A community is a union of chains (Lemma~\ref{lem:union}), and it becomes a list of label ranges rather than a list of vertex ids.
%
If the graph is kept under its original labels, a permutation of $4n$ bytes accompanies the index.
%
If the graph is stored in the index's order, nothing is added.
\begin{example}\label{ex:labels}
Consider Figure~\ref{fig:layout}.
%
The four chains own the label intervals $[0,3)$, $[3,8)$, $[8,14)$ and $[14,15)$, and the bitmap has ones at labels $0$, $3$, $8$ and $14$.
%
The vertex $u_1$ has label $8$.
%
Three ones lie at or before it, and its chain is $3-1=2$.
\end{example}

\subsection{Chain order}\label{sec:order}
The number of ranges an answer needs depends on how the chains are ordered.
%
Fix any preorder numbering $\mathrm{pre}_s$ of each forest $\Tree{s}$.
%
Give chain $c$ the key $\mathrm{key}(c)=(\mathrm{pre}_2(\own{2}{c}),\mathrm{pre}_3(\own{3}{c}),\dots)$, the preorder positions of its own nodes size by size.
%
Chains are ranked in lexicographic key order.
%
The chain of the vertices in no edge has an empty key and is placed last by convention.
%
The order has one guarantee at size $2$.
\begin{theorem}[single ranges at size 2]\label{thm:onerange}
Under the lexicographic order, the labels of every $\Nuc{2}{k}$ form one interval.
\end{theorem}
\begin{proof}
An $\Nuc{2}{k}$ is the vertex set of the subtree of $\Tree{2}$ rooted at some node $X$.
%
A chain lies in it if and only if $\own{2}{c}$ is in that subtree.
%
The preorder numbers of a subtree form an interval, and the chains whose first key component lies in an interval are contiguous in lexicographic order.
%
Contiguous ranks are contiguous labels.
\end{proof}
\begin{example}\label{ex:order}
Consider Figure~\ref{fig:example} with the preorder root $=0$, $A=1$, $\{x,u_1,\dots,w_3\}=2$ of $\Tree{2}$.
%
The keys start with $0$ for $\{b_i\}$, $1$ for $\{a_i\}$ and $2$ for $\{u,w\}$ and $\{x\}$.
%
The shorter key of $\{u,w\}$ ranks before the longer key of $\{x\}$.
%
The ranks are $\{b_i\}<\{a_i\}<\{u,w\}<\{x\}$, the labels of Figure~\ref{fig:layout}.
%
The $(2,4)$-nucleus containing $x$, $\{x,u_1,\dots,w_3\}$, is the interval $[8,15)$, and the $(2,3)$-nucleus is $[0,15)$.
\end{example}
\begin{corollary}\label{cor:kcore}
The size-$2$ layer of the index is a connected-$k$-core community index in which every community is one label range.
\end{corollary}
Indexes built for $k$-core community search report a subtree of a core tree or a list of vertices \cite{Barbieri2015ShellStruct,Fang2016CLtree}.
%
Here the same answer is the two ends of one range, and the same labels serve every larger size.

\subsection{Run arrays and entry points}\label{sec:runs}
The \textit{depth-first array} of size $s$ lists the chains that are active at size $s$ in a depth-first traversal of $\Tree{s}$.
%
At each node, the chains whose own node it is and its child subtrees are listed together, in increasing order of the smallest rank they contain, a chain counting as its own rank.
%
The roots are listed in the same order.
%
The chains of any subtree are then one contiguous segment of the array, and the first chain of a node's segment is the smallest rank in its subtree.
%
For $s=2$ the array is the rank order itself.
%
Along the array, two consecutive chains with consecutive ranks own adjacent label intervals.
%
We merge maximal sequences of such chains into \textit{runs} and store the array as the list of run endpoints $(lo,hi)$.
%
Every node $X$ stores one \textit{entry}, the pair $(r_X,v_X)$: the index of the run that holds the first label of the segment of $X$, and that label.
%
A sentinel entry $(R,hi(R-1))$ follows the last node, where $R$ is the number of runs.
\begin{theorem}[retrieval]\label{thm:retrieval}
Let $Y$ be the node that follows the subtree of $X$ in preorder, or the sentinel if there is none.
%
If $r_X=r_Y$, the labels of the community of $X$ are the one range $[v_X,v_Y)$.
%
Otherwise they are the first range $[v_X,hi(r_X))$, the whole runs $r_X+1,\dots,r_Y-1$, and the last range $[lo(r_Y),v_Y)$, which is empty when $Y$ is the sentinel.
%
The number of nonempty ranges equals the number of runs that meet the segment, and at $s=2$ it is one.
\end{theorem}
\begin{proof}
The segment of $X$ runs from the position of its entry to the position of the entry of $Y$.
%
Runs are maximal over the whole array.
%
Inside the segment a run can be cut only at the two ends of the segment.
%
The run holding the first position contributes its labels from $v_X$ on, and the run holding the last position contributes up to $v_Y$.
%
If the segment ends exactly at a run boundary, the last range is empty.
%
The runs strictly between are whole.
%
Each run that meets the segment gives one range.
%
At $s=2$ the whole array is one run, since the array is the rank order and Theorem~\ref{thm:onerange} makes every subtree one interval of labels.
\end{proof}
\begin{example}\label{ex:runs}
Consider the layer $s=3$ in Figure~\ref{fig:layout}.
%
The preorder is $N$, then $A$, and the depth-first array over chains is $\{b_i\}$, $\{u,w\}$, $\{x\}$, $\{a_i\}$.
%
The chains $\{u,w\}$ and $\{x\}$ have consecutive ranks and merge into one run, and the array is stored as the three runs $[0,3)$, $[8,15)$, $[3,8)$.
%
The entries are $N\to(0,0)$, $A\to(2,3)$ and the sentinel $(3,8)$.
%
The community of $b_1$ at $(3,3)$ is $N$, whose successor is $A$, with $r_N=0$ and $r_A=2$.
%
The answer is the first range $[0,3)$, the whole run $[8,15)$ and the empty last range $[lo(2),3)=[3,3)$: two ranges for ten vertices.
%
The community $A$ at $(3,6)$ has the sentinel as successor: $r_A=2\ne3$, so the answer is the first range $[3,hi(2))=[3,8)$, no whole run and an empty last range.
\end{example}
The community query of Section~\ref{sec:queries} first climbs to the right node.
%
The node's community is then located with two entry reads and reported as one range per run it meets.

\subsection{Values}\label{sec:values}
% 2026-09-21 the certified tail moved here from the former hierarchy section (house order: the theorem where it is used).
Core values are constant on a chain (Theorem~\ref{thm:chains}).
%
The values of a chain are one number per size from $2$ to its clique number $\omega$.
%
Most of them need not be stored.
%
A vertex lies in $\binomc{\omg{v}-1}{s-1}$ $s$-cliques inside its largest clique alone, and its value can never fall below that number, which we call its \textit{floor} at size $s$.
%
The next theorem says that once the value reaches this floor at some size, it stays on the floor at every larger size.
\begin{theorem}[floor and certified tail]\label{thm:tail}
Let $2\le s\le\omg{v}$ and $f_s(v)=\binomc{\omg{v}-1}{s-1}$.
\begin{enumerate}[label=(\alph*),leftmargin=*,nosep]
\item $\core{s}{v}\ge f_s(v)$.
\item If $\core{s}{v}=f_s(v)$ and $s<\omg{v}$, then $\core{s+1}{v}=f_{s+1}(v)$.
\end{enumerate}
\end{theorem}
\begin{proof}
(a) A largest clique $M$ containing $v$ has $\omg{v}$ vertices.
%
Every vertex of $M$ lies in $\binomc{\omg{v}-1}{s-1}$ $s$-cliques of $M$, and $M$ is connected through them.
%
Hence $M$ lies in an $\Nuc{s}{f_s(v)}$.

(b) We use the Kruskal--Katona theorem in Lov\'asz's form \cite{kruskal1963,katona1968,lovasz1979}.
%
For $k\ge1$ let $t\ge s$ be the real number with $\binomc{t}{s}=k$, the binomial coefficient extended to real $t$ by its falling factorial.
%
Then any $k$ distinct $s$-element sets have at least $g_s(k)=\binomc{t}{s-1}$ distinct $(s-1)$-element subsets.
%
Both $t\mapsto\binomc{t}{s}$ and $t\mapsto\binomc{t}{s-1}$ increase strictly for $t\ge s-1$.
%
Let $Q$ be the $\Nuc{s+1}{\core{s+1}{v}}$ containing $v$ and take any $u\in Q$.
%
The vertex $u$ lies in at least $\core{s+1}{v}$ cliques of $s+1$ vertices inside $Q$; removing $u$ from them leaves as many distinct $s$-element sets of neighbours of $u$ in $Q$.
%
These sets have at least $g_s(\core{s+1}{v})$ distinct $(s-1)$-element subsets, and each such subset together with $u$ is an $s$-clique of $Q$ through $u$.
%
Every vertex of $Q$ therefore lies in at least $g_s(\core{s+1}{v})$ $s$-cliques of $Q$.
%
$Q$ is connected through them, because two $(s+1)$-cliques that share a vertex contain $s$-cliques that share it.
%
Hence $\core{s}{v}\ge g_s(\core{s+1}{v})$.
%
Now suppose $\core{s+1}{v}>f_{s+1}(v)=\binomc{\omg{v}-1}{s}$.
%
Then the $t$ with $\binomc{t}{s}=\core{s+1}{v}$ exceeds $\omg{v}-1\ge s$, so $\core{s}{v}\ge g_s(\core{s+1}{v})=\binomc{t}{s-1}>\binomc{\omg{v}-1}{s-1}=f_s(v)$, contradicting $\core{s}{v}=f_s(v)$.
%
Since $\core{s+1}{v}\ge f_{s+1}(v)$ by (a), equality holds.
\end{proof}
We write $\sig{v}$, the \textit{certification point}, for the smallest size at which the value equals the floor, and we set $\sig{v}=\omg{v}+1$ when no size does.
%
For $s\ge\sig{v}$ the value is $\binomc{\omg{v}-1}{s-1}$ and is not stored.
%
The values for $2\le s<\sig{v}$ are the \textit{residue} of $v$.
\begin{example}\label{ex:tail}
Consider Figure~\ref{fig:example}.
%
The vertices of $A$ have $\omega=5$ and $\core{2}{a_i}=4=\binom{4}{1}$, and $\sig{a_i}=2$.
%
No value of $A$ is stored: $\core{3}{a_i}=\binom{4}{2}=6$, $\core{4}{a_i}=\binom{4}{3}=4$ and $\core{5}{a_i}=\binom{4}{4}=1$ are read from the floor.
%
The vertices $b_i$ have $\omega=4$ and $\core{2}{b_i}=3=\binom{3}{1}$, and $\sig{b_i}=2$ as well.
%
The vertex $x$ has $\omega=4$, $\core{2}{x}=4>\binom{3}{1}$ and $\core{3}{x}=3=\binom{3}{2}$.
%
Its certification point is $3$, and its residue is the single value $\core{2}{x}=4$.
%
The six vertices $u_1,\dots,w_3$ have $\omega=3$, $\core{2}{u_i}=4>\binom{2}{1}=2$ and $\core{3}{u_i}=3>\binom{2}{2}=1$.
%
No size reaches the floor, their certification point is $4=\omega+1$, and both values are residues.
%
Stored per chain, the whole graph keeps three values: one for the chain $\{x\}$ and two for the chain $\{u_1,\dots,w_3\}$.
\end{example}
\begin{remark}
On real graphs the tail carries most values.
%
\dblp has $1{,}246{,}933$ pairs $(v,s)$ with $\core{s}{v}\ge1$; kept per vertex, $84{,}010$ residues remain in total, and kept per chain, $9{,}737$.
%
\webuk has $23.5$ million such pairs and stores one residue.
\end{remark}
The record of a chain therefore holds $\omega$, $\sigma$, the residue, and the trajectory of the chain (Definition~\ref{def:chain}), from which the community query starts.
%
Every value at or past $\sigma$ is read from a table of binomial coefficients, and for $s>\omega$ it is $0$.
%
Node tops and residues are stored at a per-size byte width.
%
For every size, the widest value of that size decides whether its values take $1$, $2$, $4$ or $8$ bytes.
%
Fields are aligned to their width, and a query reads the width of the queried size once.
%
A value below $2^{32}$ is an integer of $1$, $2$ or $4$ bytes.
%
A larger value is an $8$-byte double, exact below $2^{53}$.
%
This covers every value on our graphs.
%
The decomposition itself is computed with exact integers.

\subsection{Size}\label{sec:size}
Let $C$ be the number of chains and $M$ the number of canonical nodes over all sizes.
%
Let $P=\sum_c(\omega(c)-1)$ be the number of (chain, size) pairs and $R$ the number of runs over all sizes.
%
The map takes $n/8$ bytes for the bitmap, $n/16$ for the directory and $4C$ for the first labels.
%
The chain records take $4C$ bytes for $\omega$ and $\sigma$, $4P$ for the trajectories, and the residues.
%
The trees take $20M$ bytes, a parent, a subtree size, an entry of two words and a jump pointer per node, plus $8R$ for the runs and the tops.
%
\strees takes $8\sum_v(\omg{v}-1)$ bytes for its arrays and own-node pointers, a value for each of these (vertex, size) pairs, and $12M$ bytes for its nodes plus the tops.
%
Per node the trees of \chainidx cost more, $20$ bytes against $12$.
%
The saving is on the vertex axis, $\sum_v(\omg{v}-1)$ against $P$, and on the values, the residues against a value per pair.
%
On the running example the index stores three values, and \strees stores $44$, one per (vertex, size) pair.
% trace 2026-09-21: active pairs = a_i: sizes 2..5 = 4 each, 20; b_i: sizes 2..4 = 3 each, 9; x: 3; u_i,w_i: sizes 2,3 = 2 each, 12; total 44.
