\section{The Clique Quotient}\label{sec:quotient}
% Drafted per SigmodPlus.md 239: class/pattern definitions, Class Containment lemma,
% Interchangeability theorem, pattern multiplicity, compression; example throughout.
Algorithm~\ref{alg:explicit} pays once per scored clique, so its cost is fixed by how many scored cliques exist.
%
This section asks when two scored cliques are genuinely different, and shows that on real graphs most of them are not.
%
We first group vertices, then group cliques, and finally quantify the collapse.

\stitle{Classes.}
For a vertex $v$, let $\mathcal{M}(v)=\{M\in\mathcal{M}(G): v\in M\}$ denote the maximal cliques that contain it.

\begin{definition}[Class]\label{def:class}
A \emph{class} of $G$ is a maximal set of vertices that all have the same maximal-clique membership: $u$ and $v$ belong to one class if and only if $\mathcal{M}(u)=\mathcal{M}(v)$.
\end{definition}

Classes partition $V$, and the number of classes never exceeds $n$.
%
In Figure~\ref{fig:example}, the classes are $A=\{v_1,\dots,v_4\}$ inside $M_1$ only; $B=\{v_5,v_6\}$ shared by $M_1$ and $M_2$; $C=\{v_7,v_8,v_9\}$ shared by $M_2$ and $M_3$; and $D=\{v_{10}\}$ inside $M_3$ only.
%
The vertex colors of the figure encode exactly this partition.
%
Classes sit cleanly inside maximal cliques, which the following lemma records.

\begin{lemma}[Class Containment]\label{lem:containment}
Let $c$ be a class and $M$ a maximal clique.
%
Then either every vertex of $c$ belongs to $M$, or none does.
\end{lemma}

\begin{proof}
Suppose $u\in c$ lies in $M$, and let $v\in c$ be any other vertex of the class.
%
Then $M\in\mathcal{M}(u)=\mathcal{M}(v)$, so $v\in M$.
\end{proof}

\stitle{Interchangeability.}
Vertices of one class are not merely similar; they are indistinguishable by the graph itself.

\begin{theorem}[Interchangeability]\label{thm:interchange}
Let $u\neq v$ be vertices of one class.
%
Then the transposition that swaps $u$ and $v$ and fixes every other vertex is an automorphism of $G$.
%
Consequently, for every $s$ and every $r$-clique $R$ with $u\in R$ and $v\notin R$, $\kappa_s(R)=\kappa_s\bigl(R\setminus\{u\}\cup\{v\}\bigr)$.
\end{theorem}

\begin{proof}
Let $M$ be any maximal clique containing $u$; since $\mathcal{M}(u)=\mathcal{M}(v)$, it also contains $v$, so $u$ and $v$ are adjacent.
%
Let $w$ be any neighbor of $u$ other than $v$.
%
The edge $\{u,w\}$ lies in some maximal clique $M'$, and $M'\in\mathcal{M}(u)=\mathcal{M}(v)$ places $v$ in $M'$, so $w$ is also a neighbor of $v$.
%
Hence $N[u]\subseteq N[v]$, and by symmetry $N[u]=N[v]$.
%
A transposition of two adjacent vertices with equal closed neighborhoods preserves adjacency, so it is an automorphism.
%
Core numbers depend only on the isomorphism type of $G$, so every automorphism preserves $\kappa_s$.
%
Applying the transposition to $R$ yields $R\setminus\{u\}\cup\{v\}$, which proves the claim.
\end{proof}

In Figure~\ref{fig:example}, swapping $v_1$ and $v_2$ maps the graph onto itself, so every core number that mentions $v_1$ has a twin that mentions $v_2$.

\stitle{Patterns.}
Interchangeability licenses forgetting which vertices a scored clique uses.

\begin{definition}[Pattern]\label{def:pattern}
The \emph{pattern} of an $r$-clique $R$ is the multiset of classes of its vertices, written $P=\{c_1^{\,b_1},\dots,c_t^{\,b_t}\}$ with $b_1+\dots+b_t=r$, where $R$ takes $b_i$ vertices from class $c_i$.
\end{definition}

\begin{corollary}\label{cor:pattern}
Two $r$-cliques with the same pattern have the same core number at every $s$.
\end{corollary}

\begin{proof}
Permuting vertices within one class is a product of same-class transpositions, each an automorphism by Theorem~\ref{thm:interchange}, so it is an automorphism.
%
For two cliques with one pattern, some within-class permutation maps the first onto the second.
%
Automorphisms preserve $\kappa_s$, which proves the claim.
\end{proof}

Core numbers are therefore functions of patterns, and we write $\kappa_s(P)$ for the shared value.
%
The peel of Algorithm~\ref{alg:explicit} can run on patterns directly, provided each pattern knows how many cliques it stands for.

\begin{lemma}[Pattern Multiplicity]\label{lem:mult}
Let $P=\{c_1^{\,b_1},\dots,c_t^{\,b_t}\}$ be the pattern of at least one $r$-clique, and let $n_i=|c_i|$.
%
Then the number of $r$-cliques with pattern $P$ is exactly $\prod_{i=1}^{t}\binom{n_i}{b_i}$.
\end{lemma}

\begin{proof}
Some $r$-clique with pattern $P$ lies in a maximal clique $M$, so each $c_i$ intersects $M$, and Lemma~\ref{lem:containment} puts each $c_i$ wholly inside $M$.
%
All vertices of $c_1\cup\dots\cup c_t$ are then pairwise adjacent, so every choice of $b_i$ vertices from each $c_i$ forms an $r$-clique, and each such clique has pattern $P$.
%
The number of choices is $\prod_i\binom{n_i}{b_i}$.
\end{proof}

\stitle{The collapse, quantified.}
Summing Lemma~\ref{lem:mult} over the realized patterns counts all scored cliques.
%
The ratio of cliques to patterns is therefore the average multiplicity:
\begin{equation}\label{eq:compression}
\frac{\#r\text{-cliques}}{\#\text{patterns}}
\;=\;
\operatorname*{avg}_{P}\ \prod_{i}\binom{n_i}{b_i}.
\end{equation}
The ratio is driven by class sizes: one class of hundreds of vertices contributes binomials that dwarf any number of singleton classes.

\begin{example}\label{exa:patterns}
Consider the graph $G$ in Figure~\ref{fig:example} at $r=3$.
%
Its $33$ triangles realize the $7$ patterns of Table~\ref{tab:patterns}; for instance, all $12$ triangles with pattern $\{A^2B\}$ share the core number computed once in Example~\ref{exa:running}.
%
Across the whole plane $r\in[3,5]$, $s\le8$, the $61$ scored cliques of $G$ realize $16$ patterns.
\end{example}

On real graphs the collapse is far larger.
%
On \pkustk, a finite-element mesh, $94{,}893$ vertices form $31{,}872$ classes, and the $2.1\times10^{12}$ $5$-cliques of \webit{} realize $1.3$ million patterns, a collapse of six orders of magnitude.
%
Section~\ref{sec:index} returns to Equation~\eqref{eq:compression} as a predictor: the average is computable before any decomposition, and it decides in advance whether the quotient is worth building.

Patterns are now the unit of peeling.
%
The peel still needs their supports, and obtaining those without enumerating witnesses is the subject of the next section.
